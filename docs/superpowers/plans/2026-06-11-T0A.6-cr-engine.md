### Task 2: cr_engine — core logic

**Files:** `lib/ezagent_plugin_cr/cr_engine.ex` + test

```elixir
defmodule EzagentPluginCr.CrEngine do
  @moduledoc "CR full-publish engine. One active CR per tenant."

  alias EzagentPluginContent.Tenant.{TenantRuntime, TenantConfig}
  alias Ezagent.Socialware.ConfigStore

  @spec ensure_active_cr(String.t()) :: {:ok, map()} | {:error, term()}
  def ensure_active_cr(tid) do
    case TenantConfig.read_cr(tid, active_cr_id(tid)) do
      {:ok, cr} when cr["status"] == "open" -> {:ok, cr}
      _ ->
        cr_id = "cr-#{Date.utc_today()}-#{System.unique_integer([:positive]) |> rem(1000)}"
        cr = %{"cr_id" => cr_id, "tenant_id" => tid, "status" => "open",
          "created_by" => "system://cr-engine", "created_at" => DateTime.utc_now() |> DateTime.to_iso8601()}
        write_cr(tid, cr_id, cr)
    end
  end

  @spec publish(String.t()) :: {:ok, map()} | {:error, term()}
  def publish(tid) do
    with {:ok, cr} <- ensure_active_cr(tid),
         :ok <- CrLint.check(tid),
         :ok <- CrSnapshot.snapshot(tid),
         new_ver <- next_version(tid),
         :ok <- update_current(tid, new_ver) do
      published = Map.merge(cr, %{"status" => "published", "published_version" => new_ver, "published_at" => DateTime.utc_now() |> DateTime.to_iso8601()})
      write_cr(tid, cr["cr_id"], published)
      {:ok, published}
    end
  end

  def cancel(tid) do
    {:ok, cr} = ensure_active_cr(tid)
    write_cr(tid, cr["cr_id"], Map.put(cr, "status", "cancelled"))
  end

  defp active_cr_id(tid), do: "cr:#{tid}:active"
  defp next_version(tid) do
    releases = Path.join(TenantRuntime.release_path(tid), "v*") |> Path.wildcard()
    "v#{length(releases) + 1}"
  end
  defp update_current(tid, ver) do
    current = TenantRuntime.current_release_path(tid)
    target = Path.join(TenantRuntime.release_path(tid), ver)
    File.rm_f!(current); File.ln_s(target, current)
  end
  defp write_cr(tid, cr_id, cr) do
    ConfigStore.write_and_point(%{layer: "workspace", workspace_uri: "workspace://#{tid}",
      subject_uri: "entity://system/cr", key: "cr:#{tid}:#{cr_id}", body: cr,
      actor_uri: "system://cr-engine", source_turn_id: "publish"})
    |> case do {:ok, _} -> {:ok, cr}; e -> e end
  end
end
```

Test: create/publish/cancel lifecycle. Commit: `feat(cr): cr_engine — ensure_active_cr, publish, cancel`
